if ( ! defined( 'MH_ACROSTIC_POST' ) ) define( 'MH_ACROSTIC_POST', 906355 );

function mh_acrostic_is_target() {
	return is_singular( 'mab_article' ) && (int) get_queried_object_id() === (int) MH_ACROSTIC_POST;
}

/* Replace the flat clue dump with a mount point (runs before the archive shell wraps it). */
add_filter( 'the_content', function ( $content ) {
	if ( ! mh_acrostic_is_target() || ! in_the_loop() || ! is_main_query() ) return $content;
	return '<div id="mab-acrostic-root" class="mab-acr" data-mab-puzzle="1"></div>'
		. '<noscript>' . $content . '</noscript>';
}, 20 );
